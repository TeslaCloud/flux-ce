    Player:give_item(id, amount, data, inv_type)

    plugins/inventory/plugin/lib/sh_inventory.lua:358

    Give the player a certain item(s) by id.

    -- Adds 10 test items to the player's inventory and sets their name to 'Some Item'.
local success, error_text = target:give_item('test_item', 10, { name = 'Some Item' })
if !success then
  -- Notifies the player of an error.
  target:notify(error_text)
end

    • id String
    • amount Number optional, defaults to 1
    • data Map optional, defaults to nil fields to override on the created items
    • inv_type String optional, defaults to self.default_inventory or 'main_inventory'

    • String text of the error that occurred
